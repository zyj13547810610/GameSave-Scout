```powershell
Get-ChildItem docs/superpowers -Recurse -Filter *.md | ForEach-Object {
  $text = Get-Content $_.FullName -Raw
  "{0}: V0.3.2={1}, V0.3.3={2}, V0.3.4={3}" -f $_.Name,
    ([regex]::Matches($text, 'V0\.3\.2')).Count,
    ([regex]::Matches($text, 'V0\.3\.3')).Count,
    ([regex]::Matches($text, 'V0\.3\.4')).Count
}
```
